local function mk(className, props, parent)
	local inst = Instance.new(className)
	for k, v in pairs(props) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function corner(obj, r)
	return mk("UICorner", { CornerRadius = UDim.new(0, r) }, obj)
end

local function stroke(obj, color, thickness, transparency, mode)
	return mk("UIStroke", {
		Color = color,
		Thickness = thickness,
		Transparency = transparency or 0,
		ApplyStrokeMode = mode or Enum.ApplyStrokeMode.Border,
	}, obj)
end

local function textOutline(label, transparency)
	return stroke(label, Color3.fromRGB(0, 0, 0), 1, transparency or 0.45, Enum.ApplyStrokeMode.Contextual)
end

local function nseq(a, b)
	return NumberSequence.new({
		NumberSequenceKeypoint.new(0, a),
		NumberSequenceKeypoint.new(1, b),
	})
end

local function gradient(obj, transparency, rotation)
	local g = mk("UIGradient", {}, obj)
	if transparency then
		g.Transparency = transparency
	end
	if rotation then
		g.Rotation = rotation
	end
	return g
end

local function tween(obj, props, time)
	TweenService:Create(obj, TweenInfo.new(time or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

local function clamp(v, lo, hi)
	if v < lo then
		return lo
	elseif v > hi then
		return hi
	end
	return v
end

-- flat field look. No UIGradient on text objects: a UIGradient tints the
-- text itself in Roblox and made the values invisible.
local function styleField(obj, radius)
	obj.BorderSizePixel = 0
	obj.BackgroundColor3 = C.field
	corner(obj, radius)
	local s = stroke(obj, C.line, 1, 0.80)
	return s
end

--==============================================================
-- NUMBER FORMAT (1000 -> 1k / 2500000 -> 2.5m)
--==============================================================
local SUFFIX = { { "t", 1e12 }, { "b", 1e9 }, { "m", 1e6 }, { "k", 1e3 } }
local MULT = { k = 1e3, m = 1e6, b = 1e9, t = 1e12 }

local function formatPower(value)
	local n = math.floor(tonumber(value) or 0)
	if n < 1000 then
		return tostring(n)
	end
	for _, pair in ipairs(SUFFIX) do
		local tag, size = pair[1], pair[2]
		if n >= size then
			local q = n / size
			local text
			if q >= 100 then
				text = string.format("%d", math.floor(q))
			else
				text = string.format("%.1f", math.floor(q * 10) / 10)
				if string.find(text, "%.") then
					text = string.gsub(text, "%.?0+$", "")
				end
			end
			return text .. tag
		end
	end
end

local function parsePower(value)
	if value == nil then
		return nil
	end
	local s = string.lower(tostring(value))
	s = string.gsub(s, "[%s_,]", "")
	local number, tag = string.match(s, "^([%d%.]+)([kmbt]?)$")
	if not number then
		return nil
	end
	local n = tonumber(number)
	if not n then
		return nil
	end
	if tag ~= "" then
		n = n * MULT[tag]
	end
	return math.floor(clamp(n, 1, MAX_POWER))
end

--==============================================================
-- TIME FORMAT (3 -> "3s", 3.2 -> "3.2s", 0 -> "OFF")
--==============================================================
local function formatTime(value)
	local n = tonumber(value) or 0
	if n <= 0 then
		return "OFF"
	end
	n = math.floor(n * 10 + 0.5) / 10
	if n == math.floor(n) then
		return string.format("%d", n) .. "s"
	end
	return string.format("%.1f", n) .. "s"
end

-- nil = "not a number" (so an empty box restores the old value)
local function parseTime(value)
	local s = string.lower(tostring(value or ""))
	s = string.gsub(s, "[%s_,]", "")
	s = string.gsub(s, "sec$", "")
	s = string.gsub(s, "s$", "")
	if s == "" then
		return nil
	end
	if s == "off" or s == "no" or s == "none" then
		return 0
	end
	local n = tonumber(s)
	if not n then
		return nil
	end
	return math.floor(clamp(n, 0, MAX_TIME) * 10 + 0.5) / 10
end

local function friendlyKey(name)
	local text = string.gsub(name, "^Button", "")
	text = string.gsub(text, "^DPad", "D-")
	text = string.gsub(text, "^Keypad", "NUM")
	if text == "" then
		return "?"
	end
	if string.len(text) == 1 then
		return string.upper(text)
	end
	return text
end

local function bindPreview()
	if bind.kind == "none" then
		return "{NONE}"
	end
	return "{" .. bind.label .. "}"
end

local function refreshKeyButton()
	keyBtn.Text = bindPreview()
	if bind.kind == "none" then
		keyBtn.TextColor3 = C.dim
		clearBtn.TextColor3 = C.dim
	else
		keyBtn.TextColor3 = C.white
		clearBtn.TextColor3 = C.soft
	end
end

local function setBind(kind, code, label, save)
	bind.kind = kind
	bind.code = code
	bind.label = label
	refreshKeyButton()
	if save ~= false and queueSave then
		queueSave()
	end
end

local function inputMatchesBind(input)
	if bind.kind == "none" or capturing then
		return false
	end
	if bind.kind == "mouse" then
		local wanted = Enum.UserInputType[bind.code]
		return wanted ~= nil and input.UserInputType == wanted
	end
	if input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode.Name == bind.code then
		return true
	end
	return false
end

local function setCollapsed(state, instant)
	collapsed = state
	local parts = {
		powerLabel, powerChip, powerBox,
		keyLabel, keyBtn, clearBtn,
		autoLabel, autoBtn, timeLabel, timeBox,

	for _, part in ipairs(parts) do

	end
	minBtn.Text = state and "+" or "u{2212}"
	local targetSize = UDim2.fromOffset(PANEL_W, state and H_CLOSED or H_OPEN)
	local targetPos = UDim2.fromOffset(PAD, state and Y_START_C or Y_START)
	local targetBtn = UDim2.fromOffset(INNER, state and (BTN_H - 2) or BTN_H)
	if instant then
		panel.Size = targetSize
		startBtn.Position = targetPos
		startBtn.Size = targetBtn
	else
		tween(panel, { Size = targetSize }, 0.18)
		tween(startBtn, { Position = targetPos, Size = targetBtn }, 0.18)
	end
	if queueSave then
		queueSave()
	end
end

--==============================================================
-- LAGGER (with auto-stop timer)
--==============================================================
local stopLagger

local function startLagger(source)
	if running then
		return
	end
	local parsed = parsePower(powerBox.Text)
	if parsed then
		power = parsed
	end
	powerBox.Text = formatPower(power)
	powerChip.Text = formatPower(power)

	if not net then
		warn("[ARAB PACKET LAGGER] Remote not ready - wait a moment and press START again.")
		return
	end

	running = true
	autoStarted = (source == "auto")
	runToken = runToken + 1
	runClock = os.clock()
	local token = runToken

	setRunningVisuals(true)

	local packets = table.create(power, function() end)
	task.spawn(function()
		while running and token == runToken do
			for _ = 1, 3 do
				pcall(function()
					net:FireServer(packets)
				end)
			end
			task.wait(0.01)
		end
	end)

	-- auto-stop + live countdown on the STOP button
	task.spawn(function()
		while running and token == runToken do
			if timeLimit > 0 then
				local left = timeLimit - (os.clock() - runClock)
				if left <= 0 then
					if running and token == runToken then
						stopLagger()
					end
					break
				end
				startBtn.Text = string.format("STOP %.1fs", left)
				task.wait(0.08)
			else
				startBtn.Text = "STOP"
				task.wait(0.3)
			end
		end
	end)

	if queueSave then
		queueSave()
	end
end

function stopLagger()
	if not running then
		return
	end
	running = false
	runToken = runToken + 1
	autoStarted = false
	setRunningVisuals(false)
	if queueSave then
		queueSave()
	end
end

local function toggleLagger()
	if running then
		-- stopping while holding is enough: auto only re-arms on the next grab
		stopLagger()
	else
		startLagger("manual")
	end
end

startBtn.MouseButton1Click:Connect(toggleLagger)

local function beginCapture()
	if capturing then
		return
	end
	capturing = true
	keyBtn.Text = "{PRESS}"
	keyBtn.TextColor3 = C.white
	tween(keyStroke, { Thickness = 1.6 }, 0.12)
end

local function endCapture()
	capturing = false
	tween(keyStroke, { Thickness = 1 }, 0.12)
	refreshKeyButton()
end

local function overKeyButton(position)
	local a = keyBtn.AbsolutePosition
	local s = keyBtn.AbsoluteSize
	return position.X >= a.X and position.X <= a.X + s.X
		and position.Y >= a.Y and position.Y <= a.Y + s.Y
end

local function handleCapture(input)
	local kind = input.UserInputType

	-- clicking outside the keybind box cancels the capture
	if kind == Enum.UserInputType.MouseButton1
		or kind == Enum.UserInputType.Touch
		or kind == Enum.UserInputType.MouseButton2
		or kind == Enum.UserInputType.MouseButton3
	then
		if not overKeyButton(input.Position) then
			endCapture()
			return
		end
	end

	if kind == Enum.UserInputType.Keyboard or string.find(kind.Name, "Gamepad") then
		local key = input.KeyCode
		if key == Enum.KeyCode.Unknown then
			return
		end
		if key == Enum.KeyCode.Escape or key == Enum.KeyCode.ButtonB then
			endCapture()
			return
		end
		endCapture()
		setBind("key", key.Name, friendlyKey(key.Name))
	elseif kind == Enum.UserInputType.MouseButton2 then
		endCapture()
		setBind("mouse", "MouseButton2", "RMB")
	elseif kind == Enum.UserInputType.MouseButton3 then
		endCapture()
		setBind("mouse", "MouseButton3", "MMB")
	end
end

keyBtn.MouseButton1Click:Connect(function()
	if capturing then
		endCapture() -- click again to cancel
	else
		beginCapture()
	end
end)

clearBtn.MouseButton1Click:Connect(function()
	endCapture()
	setBind("none", "None", "NONE")
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if capturing then
		handleCapture(input)
		return
	end
	-- never fire while a text box (ours or the chat) has focus
	if UserInputService:GetFocusedTextBox() then
		return

		toggleLagger()
	end
end)

--==============================================================
-- AUTO BRAINROT (ported from the Havoc logic)
-- Holding a brainrot -> start
-- Releasing it  -> stop (0.35s debounce so jumps don't flicker)
--==============================================================
-- Detects "I am carrying a brainrot".
-- 1) something in the character is literally named brainrot
-- 2) the game slowed your walk speed down (carrying mechanic).
-- We compare against the highest speed seen recently instead of a hard
-- "25" number, so it works in games with any base speed and never fires
-- just because the game walks slow.
local peakWalk = 0

local function holdingBrainrot(dt)
	local char = player.Character
	if not char then
		return false
	end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then
		return false
	end

	for _, item in ipairs(char:GetChildren()) do
		if string.find(string.lower(item.Name), "brainrot") then
			return true
		end
	end

	local ws = hum.WalkSpeed
	peakWalk = math.max(ws, peakWalk - (dt or 0) * 1.5) -- slowly forget old speeds
	if peakWalk >= 12 and ws > 0 and ws < peakWalk * 0.75 then
		return true
	end

	return false
end

task.spawn(function()
	local offTimer = 0
	local wasHeld = false

	while true do
		local dt = RunService.Heartbeat:Wait()

		if not autoBrainrot then
			offTimer = 0
			wasHeld = false
			brainrotHeld = false
			if running and autoStarted then
				stopLagger()
			end
		else
			local held = holdingBrainrot(dt)
			brainrotHeld = held

			if held then
				offTimer = 0
				if not wasHeld then
					wasHeld = true
					if not running then
						startLagger("auto")
					end
				end
			elseif wasHeld then
				offTimer = offTimer + dt
				if offTimer >= BRAINROT_OFF_DELAY then
					wasHeld = false
					offTimer = 0
					-- dropping the brainrot always ends the run while AUTO is on
					if running then
						stopLagger()
					end
				end
			else
				offTimer = 0
			end
		end
	end
end)

local saveToken = 0

function saveConfig()
	if type(writefile) ~= "function" then
		return
	end
	local ok, data = pcall(function()
		return HttpService:JSONEncode({
			version = 4,
			name = TITLE_TEXT,
			power = power,
			powerText = powerBox.Text,
			keybind = { kind = bind.kind, code = bind.code, label = bind.label },
			autoBrainrot = autoBrainrot,
			timeLimit = timeLimit,
			position = {
				x = math.floor(panel.Position.X.Offset),
				y = math.floor(panel.Position.Y.Offset),
			},
			collapsed = collapsed,
		})
	end)
	if ok and type(data) == "string" then
		pcall(writefile, CFG_FILE, data)
	end
end

function queueSave()
	if type(writefile) ~= "function" then
		return
	end
	saveToken = saveToken + 1
	local mine = saveToken
	task.delay(AUTOSAVE, function()
		if mine == saveToken then
			saveConfig()
		end
	end)
end

function loadConfig()
	if type(isfile) ~= "function" or type(readfile) ~= "function" then
		return false
	end
	local okFile, hasFile = pcall(isfile, CFG_FILE)
	if not okFile or not hasFile then
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(CFG_FILE))
	end)
	if not ok or type(data) ~= "table" then
		return false
	end

	local restored
	if type(data.powerText) == "string" then
		restored = parsePower(data.powerText)
	end
	if not restored and tonumber(data.power) then
		restored = parsePower(data.power)
	end
	if restored then
		power = restored
	end
	if type(data.powerText) == "string" and data.powerText ~= "" then
		powerBox.Text = data.powerText
	else
		powerBox.Text = formatPower(power)
	end
	powerChip.Text = formatPower(power)

	if type(data.keybind) == "table" then
		local kind = data.keybind.kind
		local code = data.keybind.code
		local label = data.keybind.label
		if kind == "none" then
			setBind("none", "None", "NONE", false)
		elseif kind == "mouse" and type(code) == "string" and Enum.UserInputType[code] then
			setBind("mouse", code, tostring(label or code), false)
		elseif type(code) == "string" and code ~= "" then
			local real = Enum.KeyCode[code]
			if real then
				setBind("key", code, tostring(label or friendlyKey(code)), false)
			end
		end
	end

	if data.autoBrainrot ~= nil then
		autoBrainrot = (data.autoBrainrot == true)
		refreshAuto(false)
	end

	if data.timeLimit ~= nil then
		local t = tonumber(data.timeLimit)
		if t then
			timeLimit = math.floor(clamp(t, 0, MAX_TIME))
			timeBox.Text = formatTime(timeLimit)
			timeBox.TextColor3 = (timeLimit > 0) and C.white or C.dim
		end
	end

	local pos = data.position
	if type(pos) == "table" and tonumber(pos.x) and tonumber(pos.y) then
		panel.Position = UDim2.fromOffset(math.floor(tonumber(pos.x)), math.floor(tonumber(pos.y)))
	end

	if data.collapsed == true then
		setCollapsed(true, true)
	end

	return true
end
