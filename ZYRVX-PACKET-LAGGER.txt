local function loadConfigIntoMemory()
 if not (readfile and isfile) then return end
 local ok, raw = pcall(function()
  if isfile(CONFIG_FILE) then return readfile(CONFIG_FILE) end
 end)
 if not ok or not raw or raw == "" then return end
 local ok2, data = pcall(function()
  return HttpService:JSONDecode(raw)
 end)
 if not ok2 or type(data) ~= "table" then return end

 if tonumber(data.Intensity) then
  config.Intensity = math.clamp(math.floor(tonumber(data.Intensity)), 1, 100000)
 end
 if type(data.Keybind) == "string" and data.Keybind ~= "" then
  config.Keybind = data.Keybind
 end
 if tonumber(data.PX_S) and tonumber(data.PX_O) and tonumber(data.PY_S) and tonumber(data.PY_O) then
  config.PX_S, config.PX_O = tonumber(data.PX_S), tonumber(data.PX_O)
  config.PY_S, config.PY_O = tonumber(data.PY_S), tonumber(data.PY_O)
 end
 if data.Collapsed ~= nil then
  config.Collapsed = not not data.Collapsed
 end
 if data.Running ~= nil then
  config.Running = not not data.Running
 end
end

loadConfigIntoMemory()

local function tween(obj, props, t)
 TweenService:Create(obj, TweenInfo.new(t or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props):Play()
end

local function clampIntensity(v)
 v = tonumber(v)
 if not v then return intensity end
 return math.clamp(math.floor(v), 1, 100000)
end

local function updateIntensity()
 intensity = clampIntensity(IntBox.Text)
 IntBox.Text = tostring(intensity)
end

local function setIntensity(v, silent)
 intensity = clampIntensity(v)
 IntBox.Text = tostring(intensity)
 if not silent then playSfx("Click") end
end

-- ============ KEY HELPERS ============
local function nameToKey(name)
 if not name then return Enum.KeyCode.G end
 local ok, key = pcall(function() return Enum.KeyCode[name] end)
 if ok and key then return key end
 return Enum.KeyCode.G
end

local function applyKeybind(name)
 toggleKeyName = name or "G"
 toggleKey = nameToKey(toggleKeyName)
 KeyBtn.Text = toggleKeyName
end

applyKeybind(config.Keybind)

-- ============ SAVE ============
local function saveConfig(silent)
 updateIntensity()
 local cfg = {
  Intensity = intensity,
  Keybind = toggleKeyName,
  PX_S = Main.Position.X.Scale, PX_O = Main.Position.X.Offset,
  PY_S = Main.Position.Y.Scale, PY_O = Main.Position.Y.Offset,
  Collapsed = collapsed,
  Running = lagging,
 }
 if writefile then
  local ok = pcall(function()
  writefile(CONFIG_FILE, HttpService:JSONEncode(cfg))
  end)
  if not silent then
  if ok then playSfx("Saved") else playSfx("Error") end
  end
 end
end

local function paintRunning()
 StartBtn.Text = "STOP"
 StartBtn.TextColor3 = TEXT_BLACK
 tween(StartBtn, {BackgroundColor3 = ACTIVE_WHITE}, 0.15)
 tween(StartStroke, {Color = TEXT_WHITE, Transparency = 0}, 0.15)

local function paintIdle()
 StartBtn.Text = "START"
 StartBtn.TextColor3 = TEXT_WHITE
 tween(StartBtn, {BackgroundColor3 = START_DARK}, 0.15)
 tween(StartStroke, {Color = TEXT_WHITE, Transparency = 0.35}, 0.15)
end

local function startLagging(silent)
 if lagging then return end
 if not Event then
  if not silent then playSfx("Error") end
  return
 end
 updateIntensity()
 lagging = true
 paintRunning()
 if not silent then playSfx("Start") end

 local payload = table.create(intensity, function() end)

 lagThread = task.spawn(function()
  while lagging do
  for _ = 1, 3 do
   pcall(function() Event:FireServer(payload) end)
  end
  task.wait(0.01)
  end
 end)
end

local function stopLagging(silent)
 if not lagging then return end
 lagging = false
 if lagThread then
  pcall(task.cancel, lagThread)
  lagThread = nil
 end
 paintIdle()
 if not silent then playSfx("Stop") end
end
