-- // 3 (Map power to lag amount) — ORIGINAL, untouched
local function applyPower(val)
	power = math.clamp(val, 10000, 500000)
	local t = (power - 10000) / 490000
	lagAmount = t * 0.5
end

applyPower(power)

-- // 4 (Start / stop lag) — ORIGINAL, untouched
local function startLag()
	if lagConn then lagConn:Disconnect() end

	lagConn = RunService.RenderStepped:Connect(function()
		if not activated then return end
		if lagAmount > 0 then
			local t = tick()
			while tick() - t < lagAmount do end
		end
	end)
end

local function stopLag()
	activated = false
	if lagConn then lagConn:Disconnect(); lagConn = nil end
end

-- Bridge the supplied Auto Brainrot logic to the existing lag-core functions.
-- The original startLag() and stopLag() implementations above remain unchanged.
local function flipLag(on, isManual)
	if on then
		if not activated then
			activated = true
			startLag()
		end
	else
		if activated then stopLag() end
	end
	active = activated
	if isManual then manualOverride = true end
	setLagVisual(active)
end

local function toggle()
	flipLag(not active, true)
end

-- AUTO BRAINROT switch only enables/disables automatic speed detection.

	setAutoVisual(cfg.autoBrainrot)

	toggle()
	if compactActivate then
		compactActivate.Text = activateBtn.Text
		compactActivate.BackgroundColor3 = activateBtn.BackgroundColor3
	end
end)

-- Supplied Auto Brainrot behavior: slow character = lag on, fast character = lag off.
RunService.Heartbeat:Connect(function()
	if not cfg.autoBrainrot then
		if brainrotMode then
			brainrotMode = false
			lastBrainrotState = false
			if active then flipLag(false, false) end
		end
		if compactActivate then
			compactActivate.Text = activateBtn.Text
			compactActivate.BackgroundColor3 = activateBtn.BackgroundColor3
		end
		return
	end

	local char = lp.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end

	local currentSpeed = hum.WalkSpeed
	if currentSpeed < 25 then
		if not lastBrainrotState then
			brainrotMode = true
			lastBrainrotState = true
			manualOverride = false
			if not active then flipLag(true, false) end
		end
	else
		if lastBrainrotState then
			brainrotMode = false
			lastBrainrotState = false
			manualOverride = false
			if active then flipLag(false, false) end
		end
	end
	if compactActivate then
		compactActivate.Text = activateBtn.Text
		compactActivate.BackgroundColor3 = activateBtn.BackgroundColor3
	end
end)

setAutoVisual(cfg.autoBrainrot)
setLagVisual(active)

-- // 16 (Input listener)
UserInputService.InputBegan:Connect(function(input, gpe)
	if gpe then return end
	if waitingForKey then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			keybind = input.KeyCode
			kbBtn.Text = keybind.Name
			kbBtn.TextColor3 = Color3.fromRGB(245, 245, 245)
			waitingForKey = false
		end
		return
	end
	if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == keybind then
		-- E (or the selected key) toggles the bypass activation.
		flipLag(not active, true)
		if compactActivate then
			compactActivate.Text = activateBtn.Text
			compactActivate.BackgroundColor3 = activateBtn.BackgroundColor3
		end
	end
end)

	task.wait(1)
	if activated then
		stopLag()
		activated = true
		startLag()

		setLagVisual(true)
		if compactActivate then
			compactActivate.Text = activateBtn.Text
			compactActivate.BackgroundColor3 = activateBtn.BackgroundColor3
		end
	end
end)
